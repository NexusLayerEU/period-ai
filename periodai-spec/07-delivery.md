# 07 — Delivery Channels

## Base Delivery Interface

```python
# app/delivery/base.py

class BaseDelivery(ABC):
    @abstractmethod
    async def send(
        self,
        run: ReportRun,
        template: ReportTemplate,
        html_output: str,
        channel_config: dict
    ) -> DeliveryResult:
        ...

@dataclass
class DeliveryResult:
    success: bool
    channel: str
    sent_at: datetime | None = None
    error: str | None = None
    metadata: dict = field(default_factory=dict)
```

# app/delivery/email.py
import sendgrid
from sendgrid.helpers.mail import Mail, Attachment, FileContent, FileName, FileType

class EmailDelivery(BaseDelivery):
    async def send(self, run, template, html_output, channel_config):
        sg = sendgrid.SendGridAPIClient(api_key=settings.SENDGRID_API_KEY)
        
        subject = self.render_subject(
            channel_config["subject"],
            report_name=template.name,
            report_date=datetime.utcnow().strftime("%B %d, %Y")
        )
        
        message = Mail(
            from_email=settings.EMAIL_FROM,
            to_emails=channel_config["recipients"],
            subject=subject,
            html_content=self.wrap_html_for_email(html_output)
        )
        
        if channel_config.get("include_pdf") and run.output_pdf_url:
            # Attach PDF
            pdf_bytes = await fetch_pdf(run.output_pdf_url)
            attachment = Attachment(
                FileContent(base64.b64encode(pdf_bytes).decode()),
                FileName(f"{template.name}_{run.id}.pdf"),
                FileType("application/pdf")
            )
            message.attachment = attachment
        
        response = sg.send(message)
        
        return DeliveryResult(
            success=response.status_code in (200, 202),
            channel="email",
            sent_at=datetime.utcnow(),
            metadata={"status_code": response.status_code, "recipients": channel_config["recipients"]}
        )

    def wrap_html_for_email(self, html: str) -> str:
        """Wrap report HTML in email-safe wrapper with inline CSS."""
        return f"""

## Slack Delivery

### Channel Config

```json
{
  "type": "slack",
  "channel_id": "C0123456789",
  "bot_token_encrypted": "<AES-256 encrypted bot token>",
  "mention_users": ["U0123", "U0456"],
  "include_pdf_link": true
}
```

### Implementation

```python
# app/delivery/slack.py
from slack_sdk.web.async_client import AsyncWebClient

class SlackDelivery(BaseDelivery):
    async def send(self, run, template, html_output, channel_config):
        bot_token = decrypt(channel_config["bot_token_encrypted"])
        client = AsyncWebClient(token=bot_token)
        
        # Build Slack Block Kit message
        blocks = self.build_blocks(run, template, channel_config)
        
        response = await client.chat_postMessage(
            channel=channel_config["channel_id"],
            blocks=blocks,
            text=f"📊 {template.name} — {datetime.utcnow().strftime('%B %d, %Y')}"
        )
        
        # If PDF link available, follow up with file snippet
        if channel_config.get("include_pdf_link") and run.output_pdf_url:
            await client.chat_postMessage(
                channel=channel_config["channel_id"],
                thread_ts=response["ts"],
                text=f"📄 Download PDF: {run.output_pdf_url}"
            )
        
        return DeliveryResult(
            success=True,
            channel="slack",
            sent_at=datetime.utcnow(),
            metadata={"message_ts": response["ts"], "channel": channel_config["channel_id"]}
        )

    def build_blocks(self, run, template, config) -> list:
        """Build Slack Block Kit blocks for the report."""
        mention_str = " ".join(f"<@{u}>" for u in config.get("mention_users", []))
        
        blocks = [
            {
                "type": "header",
                "text": {"type": "plain_text", "text": f"📊 {template.name}"}
            },
            {
                "type": "context",
                "elements": [
                    {"type": "mrkdwn", "text": f"*Generated:* {datetime.utcnow().strftime('%B %d, %Y at %H:%M UTC')}"},
                    {"type": "mrkdwn", "text": f"*Duration:* {run.duration_ms / 1000:.1f}s"}
                ]
            },
            {"type": "divider"}
        ]
        
        # Add key sections as text blocks (max 3000 chars per block)
        if run.output_json:
            for section in run.output_json.get("sections", [])[:3]:  # First 3 sections in Slack
                if section["type"] == "summary" and section.get("content"):
                    blocks.append({
                        "type": "section",
                        "text": {"type": "mrkdwn", "text": f"*{section['title']}*\n{section['content'][:500]}"}
                    })
        
        # Action buttons
        actions = {
            "type": "actions",
            "elements": [
                {
                    "type": "button",
                    "text": {"type": "plain_text", "text": "View Full Report"},
                    "url": f"{settings.APP_URL}/runs/{run.id}",
                    "style": "primary"
                }
            ]
        }
        if run.output_pdf_url:
            actions["elements"].append({
                "type": "button",
                "text": {"type": "plain_text", "text": "Download PDF"},
                "url": run.output_pdf_url
            })
        blocks.append(actions)
        
        if mention_str:
            blocks.insert(0, {
                "type": "section",
                "text": {"type": "mrkdwn", "text": mention_str}
            })
        
        return blocks
```

## Webhook Delivery

### Channel Config

```json
{
  "type": "webhook",
  "url": "https://your-system.com/hooks/periodai",
  "secret": "webhook-signing-secret",
  "method": "POST",
  "headers": { "X-Custom-Header": "value" }
}
```

### Implementation

```python
# app/delivery/webhook.py
import hmac, hashlib
import httpx

class WebhookDelivery(BaseDelivery):
    async def send(self, run, template, html_output, channel_config):
        payload = {
            "event": "report.completed",
            "run_id": str(run.id),
            "template_id": str(run.template_id),
            "template_name": template.name,
            "status": run.status,
            "generated_at": run.completed_at.isoformat(),
            "duration_ms": run.duration_ms,
            "total_tokens": run.total_tokens,
            "report_url": f"{settings.APP_URL}/runs/{run.id}",
            "pdf_url": run.output_pdf_url,
            "output": run.output_json  # Full structured report data
        }
        
        payload_str = json.dumps(payload, sort_keys=True)
        
        # HMAC-SHA256 signature
        signature = hmac.new(
            channel_config["secret"].encode(),
            payload_str.encode(),
            hashlib.sha256
        ).hexdigest()
        
        headers = {
            "Content-Type": "application/json",
            "X-PeriodAI-Signature": f"sha256={signature}",
            "X-PeriodAI-Run-ID": str(run.id),
            **channel_config.get("headers", {})
        }
        
        async with httpx.AsyncClient(timeout=10.0) as client:
            response = await client.post(
                channel_config["url"],
                content=payload_str,
                headers=headers
            )
        
        return DeliveryResult(
            success=200 <= response.status_code < 300,
            channel="webhook",
            sent_at=datetime.utcnow(),
            error=None if response.is_success else f"HTTP {response.status_code}",
            metadata={"status_code": response.status_code, "url": channel_config["url"]}
        )
```

### Webhook Verification (for consumers)

Include this in the docs for webhook consumers:

```python
# How to verify PeriodAI webhook signatures
import hmac, hashlib

def verify_signature(payload_body: bytes, signature_header: str, secret: str) -> bool:
    expected = "sha256=" + hmac.new(
        secret.encode(), payload_body, hashlib.sha256
    ).hexdigest()
    return hmac.compare_digest(expected, signature_header)
```

## PDF Generation

```python
# app/delivery/pdf.py
# Uses WeasyPrint for HTML → PDF conversion

import weasyprint
import boto3  # or local storage

class PDFGenerator:
    async def generate_and_store(self, run: ReportRun, html: str) -> str:
        """Generate PDF from HTML and store it. Returns URL."""
        pdf_bytes = weasyprint.HTML(string=html).write_pdf()
        
        filename = f"reports/{run.workspace_id}/{run.template_id}/{run.id}.pdf"
        
        if settings.STORAGE_BACKEND == "s3":
            url = await self.upload_to_s3(pdf_bytes, filename)
        else:
            url = await self.store_locally(pdf_bytes, filename)
        
        return url

    async def upload_to_s3(self, pdf_bytes: bytes, key: str) -> str:
        s3 = boto3.client("s3")
        s3.put_object(
            Bucket=settings.S3_BUCKET,
            Key=key,
            Body=pdf_bytes,
            ContentType="application/pdf",
            ACL="private"
        )
        return f"https://{settings.S3_BUCKET}.s3.amazonaws.com/{key}"
```

## Delivery Orchestration

```python
# Called in pipeline Stage 6
async def deliver_all(
    run: ReportRun,
    template: ReportTemplate,
    html_output: str
):
    # Generate PDF first (used by email + storage)
    pdf_url = await pdf_generator.generate_and_store(run, html_output)
    run.output_pdf_url = pdf_url

    delivery_map = {
        "email": EmailDelivery(),
        "slack": SlackDelivery(),
        "webhook": WebhookDelivery(),
    }

    for channel_config in template.delivery_channels:
        channel_type = channel_config["type"]
        handler = delivery_map.get(channel_type)
        if not handler:
            continue

        delivery = Delivery(run_id=run.id, channel=channel_type, status="pending")
        session.add(delivery)
        await session.flush()

        try:
            result = await handler.send(run, template, html_output, channel_config)
            delivery.status = "sent" if result.success else "failed"
            delivery.sent_at = result.sent_at
            delivery.error = result.error
            delivery.metadata = result.metadata
        except Exception as e:
            delivery.status = "failed"
            delivery.error = str(e)

    await session.commit()
```
